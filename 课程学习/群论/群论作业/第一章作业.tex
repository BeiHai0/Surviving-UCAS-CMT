\setcounter{problem}{1}
\begin{problem}
证明以乘法作为“乘积”的所有正实数构成的群和以“加法”作为乘积的所有实数构成的群同构。
\end{problem}
\begin{answer}
  $\left(\mathbb{R}_+,\times \right)\equiv G,\left(\mathbb{R},+ \right)\equiv H $，构造映射 $f:H\to G $ 如下：
  \begin{equation}
    f(x) = \mathrm{e}^x,\qquad x\in H,\qquad f(x) = \mathrm{e}^x \in G.
  \end{equation}
  首先证明 $f $ 是群同态：
  \begin{equation}
    \forall x_1,x_2\in H,\qquad f(x_1+x_2)
    =\mathrm{e}^{x_1+x_2}
    =\mathrm{e}^{x_1} \mathrm{e}^{x_2}
    =f(x_1) f(x_2),
  \end{equation}
  因此 $f $ 是群同态。再证明 $f $ 是满射：
  \begin{equation}
    \forall y\in G,
    \exists \ln y\in H,\qquad
    f(\ln y) = \mathrm{e}^{\ln y} = y,
  \end{equation}
  因此 $f $ 是满射。最后证明 $f $ 是单射：设 $x_1,x_2\in H $，若 $f(x_1)=f(x_2) $，则
  \begin{equation}
    \mathrm{e}^{x_1}=\mathrm{e}^{x_2} \Longrightarrow
    x_1=x_2,
  \end{equation}
  因此 $f $ 是单射。综上，$H,G $ 同构。
\end{answer}

\setcounter{problem}{3}
\begin{problem}
证明每个元素的平方都等于恒元的群一定是阿贝尔群。
\end{problem}
\begin{answer}
  设群 $G $ 满足 $\forall g\in G,g^2=e$，则
  \begin{equation}
    g^{-1} = g,\qquad \forall g\in G.
  \end{equation}
  \begin{equation}
    \forall g_1,g_2\in G,\qquad
    g_1 g_2 = g_1^{-1} g_2^{-1} = \left(g_2 g_1 \right)^{-1} = g_2 g_1,
  \end{equation}
  因此 $G $ 是阿贝尔群。
\end{answer}

\setcounter{problem}{8}
\begin{problem}
设 $H_1$ 和 $H_2$ 是群 $G$ 的两个子群，证明 $H_1$ 和 $H_2$ 的公共元素的集合也构成群 $G$ 的子群。
\end{problem}
\begin{answer}
  设 $H_3=H_1\cap H_2 $，
  \begin{itemize}
    \item 存在恒元：由于 $H_1,H_2 $ 是群 $G $ 的子群，因此恒元 $e $ 是 $H_1,H_2 $ 的公共元素，$e\in H_3 $；
    \item 满足结合律：$H_3 $ 继承群乘法，则一定满足结合律；
    \item 封闭性：$\forall h_1,h_2\in H_3=H_1\cap H_2,h_1,h_2\in H_1,h_1 h_2 \in H_1 $。同理 $h_1 h_2\in H_2 $，因此 $h_1h_2\in H_3 $；
    \item 存在逆元：$\forall h\in H_3=H_1\cap H_2,h\in G,\exists h^{-1}\in G $。由于 $H_1,H_2 $ 都是 $G $ 的子群，因此 $h^{-1} \in H_1,H_2 $，于是 $h^{-1} \in H_3=H_1\cap H_2 $。
  \end{itemize}
\end{answer}

\begin{problem}
设集合 $G$ 包含元素数目为 $5$、$6$ 或 $7$，所有元素的平方都是恒元，证明此集合不构成群。
\end{problem}
\begin{answer}
  假设群 $G $ 所有元素的平方都是恒元，任取两个不同非恒元 $a,b\in G $ 可以构成集合 $H=\left\{e,a,b,ab \right\} $。下面证明 $H $ 是 $G $ 的子群。
  
  若 $ab=e $，则 $a=ae=a^2b=b $，这与 $a\ne b $ 矛盾；
  
  若 $ab=a $，则 $b=e $，这与 $b\ne e $ 矛盾；

  同理，肉 $ab=b $ 与 $a\ne e $ 矛盾。因此 $ab\ne e,a,b $，且显然 $H $ 对乘法封闭，而 $H $ 元素个数有限，因此 $H $ 是 $G $ 的子群。
  
  若 $G $ 的元素数目为 $5 $、$6 $ 或 $7 $，由于子群 $H $ 的阶为 $4 $，不是群 $G $ 的阶的因子，与拉格朗日定理矛盾。
\end{answer}

\setcounter{problem}{14}
\begin{problem}
准确到同构，证明九阶群 $G$ 只有两种：循环群 $C_9$ 和直乘群 $C_3\otimes C_3$。
\end{problem}
\begin{answer}
  若 $G $ 中有 9 阶元素，则此元素作为生成元可生成 $G \cong \mathrm{C}_9 $；若 $G $ 中没有 9 阶元素，则非恒元元素的阶只能为 3。取生成元 $a $ 生成子群 $H=\left\{e,a,a^2 \right\} $，再取 $b\notin H,b\in G $，其中 $a^3=b^3=e $。下面证明 $b^2 \notin bH $：

  若 $b^2 \in bH $，则 $b^2 H = b H$，也即存在 $h,h'\in H $ 使得 $b^2  h= b h'$，因此 $b=h' h^{-1} \in H $，与 $b\notin H$ 矛盾。

  于是 $H\cup bH \cup b^2H = G $，因此 $G = \left\{b^i a^j | i,j=0,1,2 \right\} $。下面证明 $G $ 是阿贝尔群：

  由封闭性，设 $ab = b^m a^n, m,n\ne 0 $。若 $m=1,n=2 $，即 $ab = ba^2 $，则 $a=b a^2 b^{-1} $，于是有
  \begin{equation}
    \begin{aligned}
      a
      &=b^3 \left(a^2 \right)^2 b^{-3}
      =b^2 \left(b a^2 b^{-1} \right) \left(b a^2 b^{-1} \right) b^{-2}
      =b^2 a^2 b^{-2}
      =b \left(b a^2 b^{-1} \right) b^{-1} \\
      &=b a b^{-1}
      =b a^4 b^{-1}
      =\left(b a^2 b^{-1} \right)\left(b a^2 b^{-1} \right)
      =a^2,
    \end{aligned}
  \end{equation}
  这与 $a\ne e $ 矛盾；
  
  若 $m=2,n=1 $ 即 $ab=b^2 a $，则 $aba^{-1}=b^2 $，于是有
  \begin{equation}
    \begin{aligned}
      b
      &=a^3 b a^{-3}
      =a^2 b^2 a^{-2}
      =a \left(a b a^{-1} \right)\left(a b a^{-1} \right) a^{-1}
      =aba^{-1}
      =b^2,
    \end{aligned}
  \end{equation}
  这与 $b\ne e $ 矛盾；
  
  若 $m=2,n=2 $，即 $ab=b^2a^2 $，则 $ab=b^2a^2=b^{-1}a^{-1}=(ab)^{-1}$，这与 $ab $ 是 3 阶元素矛盾。
  
  综上，只有 $m=1,n=1 $ 成立，即 $ba=ab $，因此 $G $ 是阿贝尔群。考虑 $H=\left\{e,a,a^2 \right\},H'=\left\{e,b,b^2 \right\} $， $H,H' $ 都是 $G $ 的不变子群，且 $H\cap H'=\left\{e \right\} $，二者元素可对易，$G $ 由 $a,b $ 生成，因此 $G\cong \mathrm{C}_3\otimes\mathrm{C}_3 $。

\end{answer}
